	\begin{frame}{Majoration}
		Avec la méthode ci-dessus:
		
		\begin{itemize}		
			\item Pour chaque ligne: $4(N-1)^3/8$ \\
			\item Pour la dernière colonne: $5(N-1)^2/4$ \\
			\item Pour la dernière ligne: $2N(N-1)/2+N/2$ \\
		\end{itemize}
		
		Soit un total de 75 mouvements pour $N=5$
	\end{frame}
	
	\begin{frame}{Minoration}
		Nombre de mélanges en moins de $k$ mouvements, pour un puzzle de taille $N$:
		$$ 1+\sum_{i=0}^k (4N-1)^i = 1 + \frac{1-(4N-1)^{k+1}}{2-4N}$$
		Si $k$ est la majoration recherchée, on a donc:
		\begin{align*}
			& \frac{N^2!}2  \geqslant 1 + \frac{1-(4N-1)^{k+1}}{2-4N} \\
			\iff & k \geqslant \frac{\log\left(\left(\frac{N^2!}2-1\right)(4N-2)-1\right)}{\log(4N-1)}
		\end{align*}
		ie, pour $N=5$, $k\geqslant 20$.
	\end{frame}

	\begin{frame}{algorithme de Heap}
		\begin{algorithm}[H]
			\DontPrintSemicolon
			\caption{Algorithme de Heap pour générer $\perm_n$}
			
			\KwData{Tableau $A$ de taille $n$}
			\KwResult{Toutes les permutations de $A$}
			
			\If{$n = 1$}{
				afficher $A$\;
			}
			\Else{
				\For{$i \leftarrow 1$ \KwTo $n$}{
					Heap$(A, n-1)$\;
					
					\eIf{$n$ est impair}{
						échanger $A[1]$ et $A[n]$\;
					}{
						échanger $A[i]$ et $A[n]$\;
					}
				}
			}
			
		\end{algorithm}
	\end{frame}
